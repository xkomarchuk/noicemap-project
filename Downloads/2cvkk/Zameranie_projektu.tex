\section{Motivácia a problém}

\begin{frame}{Hľadanie tichého miesta}
    \begin{itemize}
        \item Ľudia často strácajú množstvo času hľadaním vhodného priestoru na prácu alebo štúdium.
        \item Tento problém sa týka predovšetkým verejných a zdieľaných budov:
        \begin{itemize}
            \item Univerzitné knižnice a študovne.
            \item Kaviarne.
            \item Moderné coworkingové priestory.
        \end{itemize}
        \item \textbf{Naše riešenie:} Platforma NoiseMap pre monitorovanie a vizualizáciu hlučnosti.
    \end{itemize}
\end{frame}

\section{Zber a spracovanie údajov}

\begin{frame}{Ako projekt funguje?}
    \begin{itemize}
        \item \textbf{Zdroje meraní hluku:}
        \begin{itemize}
            \item Statické senzorové siete nainštalované priamo v budovách.
            \item Dobrovoľné crowdsourcingové merania od samotných návštevníkov.
        \end{itemize}
        \item \textbf{Spracovanie a bezpečnosť:}
        \begin{itemize}
            \item Striktné zohľadnenie ochrany súkromia (anonymizácia dát).
            \item Využitie efektívnych metód na zachytávanie a filtráciu čistého zvuku.
        \end{itemize}
    \end{itemize}
\end{frame}
